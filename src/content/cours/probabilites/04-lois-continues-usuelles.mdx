## Loi uniforme continue

<Definition titre="Loi uniforme continue">
    $$
    \forall x \in [a, b], \quad(x) = \frac{1}{b - a}
    $$

    $$
    F(x) = \frac{x - a}{b - a} \qquad (x \in [a, b])
    $$

    On note $X \sim \mathcal{U}(a, b)$.
</Definition>

<Property titre="Probabilité, espérance et variance">
    $$
    P(c < X
< d) = \frac{d - c}{b - a} \qquad (c, d) \in [a, b]
    $$

    $$
    E(X) = \frac{b + a}{2} \qquad\qquad \operatorname{Var}(X) = \frac{(a - b)^ 2}{12}
    $$
</Property>

## Loi exponentielle

<Definition titre="Loi exponentielle">
    Densité de probabilité :

    $$
    f(x) = \lambda e^{-\lambda x}
    $$

$\forall x \ge 0$, alors

    $$
    F(x) = 1 - e^{-\lambda x}
    $$

    Notation : $X \sim \mathcal{E}(\lambda)$.
</Definition>

<Property titre="Calculs de probabilités">
    $$
    \begin{aligned}
    P(a \le X \le) &= F(b) - F(a) = e^{-\lambda a} - e^{-\lambda b} \\
    P(0 \le X \le) &= F(a) = 1 - e^{-\lambda a} \\
    P(X \ge) &= 1 - P(X < b) = 1 - F(b) = e^{-\lambda b}
    \end{aligned}
    $$
</Property>

<Property titre="Espérance, variance, écart type">
    $$
    E(X) = \int x \, f(x)\,dx \qquad\Longrightarrow\qquad(X) = \frac{1}{\lambda}
    $$

    $$
    V(X) = E(X^2) - \big(E(X)\big)^2 \qquad\Longrightarrow\qquad(X) = \frac{1}{\lambda^2}
    $$

    $$
    \sigma(X) = \frac{1}{\lambda}
    $$
</Property>

<Theorem type="À retenir" titre="Loi sans mémoire">
    $$
    \begin{aligned}
    P(X \ge t + h \mid X \ge) &= \frac{P(X \ge t + h \,\cap\, X \ge t)}{P(X \ge t)} \\
    &= \frac{P(X \ge t + h)}{P(X \ge t)} = \frac{e ^ { -\lambda (t + h)}}{e ^ { -\lambda t}} \\
    &= \frac{e ^ { -\lambda t} \, e^{-\lambda h}}{e ^ { -\lambda t}} = e^{-\lambda h} \\
    &= P(X \ge)
    \end{aligned}
    $$
</Theorem>

## Loi normale

<Definition titre="Loi normale">
    2 paramètres : $\mu$ et $\sigma$ ; $\sigma > 0$ est l'écart type.

    Densité de probabilité :

    $$
    f(x) = \frac{1}{\sigma\sqrt{2\pi}} \, e^{-\frac{1}{2} \left( \frac{x - \mu}{\sigma} \right)^2}
    $$

    On note $X \sim \mathcal{N}(\mu, \sigma)$.
</Definition>

    À environ $\frac{2}{3}$ de la hauteur, la largeur de la cloche vaut $2\sigma$.

<Property titre="Espérance, variance, écart type">
    $$
    E(X) = \mu \qquad(X) = \sigma^2 \qquad \sigma(X) = \sigma
    $$
</Property>

### Fonction de répartition

<Definition titre="Fonction de répartition">
    $F(x)$ est impossible à résoudre « par formule » :

    $$
    F(x) = \frac{1}{\sigma\sqrt{2\pi}} \int_{-\infty}^{x} e^{-\frac{1}{2} \left( \frac{t - \mu}{\sigma} \right)^2} dt
    $$
</Definition>

<Remark titre="À la calculatrice">
    Stat → Dist → Norm → Ncd
</Remark>

### Loi normale centrée réduite

<Property titre="Centrer et réduire">
    Si $X \sim \mathcal{N}(\mu, \sigma)$, alors la VAR

    $$
    T = \frac{X - \mu}{\sigma}
    $$

    est la VAR centrée réduite, qui se comporte comme $T \sim \mathcal{N}(0, 1)$.
</Property>

<Property titre="Utiliser la table de Φ">
    $F_T$ est noté généralement $\Phi$ (table de $\Phi$ fournie).

    $$
    P(a \le X \le) = \Phi\left( \frac{b - \mu}{\sigma} \right) - \Phi\left( \frac{a - \mu}{\sigma} \right)
    $$
</Property>

## Fiche récapitulative

<Formulas titre="Lois continues usuelles">
    <Formula label="Uniforme 𝒰(a, b)">
        $$
        f(x) = \frac{1}{b - a}, \quad(x) = \frac{x - a}{b - a}, \quad(X) = \frac{a + b}{2}, \quad \operatorname{Var}(X)
        = \frac{(a - b)^ 2}{12}
        $$
    </Formula>
    <Formula label="Exponentielle ℰ(λ)">
        $$
        f(x) = \lambda e^{-\lambda x}, \quad(x) = 1 - e^{-\lambda x}, \quad(X) = \sigma(X) = \frac{1}{\lambda}, \quad(X)
        = \frac{1}{\lambda^2}
        $$
    </Formula>
    <Formula label="Sans mémoire (exponentielle)">
        $$
        P(X \ge t + h \mid X \ge) = P(X \ge)
        $$
    </Formula>
    <Formula label="Normale 𝒩(μ, σ)">
        $$
        f(x) = \frac{1}{\sigma\sqrt{2\pi}} e^{-\frac{1}{2}\left(\frac{x-\mu}{\sigma}\right)^2}, \quad(X) = \mu, \quad(X)
        = \sigma^2
        $$
    </Formula>
    <Formula label="Centrée réduite">
        $$
        T = \frac{X - \mu}{\sigma} \sim \mathcal{N}(0, 1)
        $$
    </Formula>
    <Formula label="Avec la table de Φ">
        $$
        P(a \le X \le) = \Phi\left(\frac{b-\mu}{\sigma}\right) - \Phi\left(\frac{a-\mu}{\sigma}\right)
$$
    </Formula>
</Formulas>
